def box_guard(box,quota=1,multiple=1,cap=True):
    cx,cy,hx,hy=map(f,box);quota=f(quota);g=f(multiple*M)
    return (quota>f(8*M) and (quota<=16 or not cap) and hx>f(1e-5) and hy>f(1e-5)
            and f(cx-hx)>=g and f(cy-hy)>=g and f(f(cx+hx)+f(cy+hy))<=f(quota-g))
def exact(box,quota=1):
    cx,cy,hx,hy=map(q,box);quota=q(quota)
    return {'actual_domain':hx>0 and hy>0 and cx-hx>=0 and cy-hy>=0 and cx+hx+cy+hy<=quota,
            'full_margin':hx>0 and hy>0 and cx-hx>=q(M) and cy-hy>=q(M) and cx+hx+cy+hy<=quota-q(M),
            'lower_x':float(cx-hx),'upper_sum':float(cx+hx+cy+hy),
            'upper_excess_over_full_margin':float(cx+hx+cy+hy-(quota-q(M)))}

# Sample near the two relevant quota boundaries. This is evidence supporting
# implementation inspection, not the universal proof supplied above.
rng=random.Random(202609253);counts={k:{'sampled':0,'admitted':0,'actual_domain_failures':0,'full_margin_failures':0} for k in [1,2]}
first_mismatch=None
for i in range(6000):
    quota=f([.001,1,4,16][i%4]);hx=f(rng.uniform(2e-5,quota*.08));hy=f(rng.uniform(2e-5,quota*.08))
    cx=f(rng.uniform(hx+3*M,max(hx+3*M,quota-hx-2*hy-4*M)))
    center_margin=(1 if i%2 else 2)*M
    cy=f(quota-center_margin-cx-hx-hy+rng.uniform(-6e-6,6e-6))
    box=[cx,cy,hx,hy];truth=exact(box,quota)
    for multiple in [1,2]:
        counts[multiple]['sampled']+=1
        if box_guard(box,quota,multiple):
            counts[multiple]['admitted']+=1
            counts[multiple]['actual_domain_failures']+=not truth['actual_domain']
            counts[multiple]['full_margin_failures']+=not truth['full_margin']
            if not truth['full_margin'] and first_mismatch is None:first_mismatch={'box':box,'quota':quota,'exact':truth}

def expression(expr,values):
    if type(expr) in (int,float):return f(expr)
    if 'input' in expr:return f(values[expr['input']])
    args=[expression(a,values) for a in expr['args']];op=expr['op']
    if op=='add':value=args[0]+args[1]
    elif op=='sub':value=args[0]-args[1]
    elif op=='mul':value=args[0]*args[1]
    elif op=='abs':value=abs(args[0])
    elif op=='min':value=min(args)
    elif op=='max':value=max(args)
    elif op=='floor':value=math.floor(args[0])
    elif op=='less':value=int(args[0]<args[1])
    elif op=='select':value=args[1] if args[0]!=0 else args[2]
    else:raise AssertionError(op)
    result=f(value);assert math.isfinite(result);return result
def actual_guards(program,quota):
    values=bindings.pack_program(program);values['quota']=f(quota);out={}
    for name in guard_names[:-1]:

    out['combined_geometry']=expression(bank['validate_geometry']['binding']['outputs']['valid'],combo)
    return out
