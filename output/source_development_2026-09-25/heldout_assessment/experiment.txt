def sha(data):
    return hashlib.sha256(data).hexdigest()

def now():
    return datetime.now(timezone.utc).isoformat()

def write(name, value):
    data = (json.dumps(value, indent=2, ensure_ascii=False, allow_nan=False) + "\n").encode("utf-8")
    (OUT / name).write_bytes(data)
    return sha(data)

def fp(value):
    return struct.unpack("<f", struct.pack("<f", value))[0]

def q(value):
    return Fraction.from_float(value)

def rectangles(program):
    cert = certificates[next(name for name, p in programs.items() if p == program)]
    result = []
    for index in cert["leaf_indices"]:
        cx,cy,hx,hy = map(q,program["boxes"][index])
        result.append((cx-hx,cx+hx,cy-hy,cy+hy))
    return result

def area(rects):
    return sum((ux-lx)*(uy-ly) for lx,ux,ly,uy in rects)

def intersection(left,right):
    return sum(max(Fraction(0),min(a[1],b[1])-max(a[0],b[0])) * max(Fraction(0),min(a[3],b[3])-max(a[2],b[2])) for a in left for b in right)

for name, program in programs.items():
    datasets = {}
    values[name] = {}
    for dataset, points in [("training",training),("holdout",queries),("outside_quota",negative)]:
        rows = []
        for index,point in enumerate(points):
            oracle = sdf.boundary_oracle(program,point)
            formula = sdf.evaluate_program(program,point)
            rows.append({"index":index,"oracle_distance":oracle["distance"],"formula_distance":formula,
                         "absolute_error":abs(formula-oracle["distance"]),
                         "covered":oracle["distance"]<=threshold,
                         "inside_or_boundary":oracle["inside_or_boundary"],
                         "coverage_boundary_ambiguous":abs(oracle["distance"]-threshold)<=band})
        values[name][dataset] = rows
        covered = sum(r["covered"] for r in rows)
        datasets[dataset] = {"count":len(rows),"covered":covered,"coverage_fraction":covered/len(rows),
            "inside_or_boundary":sum(r["inside_or_boundary"] for r in rows),
            "coverage_boundary_ambiguous":sum(r["coverage_boundary_ambiguous"] for r in rows),
            "distance_min":min(r["oracle_distance"] for r in rows),"distance_max":max(r["oracle_distance"] for r in rows),
            "mean_distance":sum(r["oracle_distance"] for r in rows)/len(rows),
            "max_formula_vs_edge_oracle_absolute_error":max(r["absolute_error"] for r in rows)}
        if dataset=="holdout":
            datasets[dataset]["covered_by_cluster"] = [sum(r["covered"] for r,label in zip(rows,labels) if label==cluster) for cluster in range(3)]
    rects, refrects = rectangles(program), rectangles(reference)
    current_area, reference_area = area(rects), area(refrects)
    overlap = intersection(rects,refrects)
    leaf_count = len(rects)
    penalty = .02*leaf_count+.005*float(current_area)
    for dataset in ["training","holdout"]:
        datasets[dataset]["count_score"] = datasets[dataset]["covered"]-penalty
    metrics[name] = {"program_hashes":sdf.program_hashes(program),"exact_certificate_admissible":certificates[name]["admissible"],
        "leaf_count":leaf_count,"node_count":program["node_count"],"revision":program["revision"],
        "union_area":float(current_area),"union_area_exact":str(current_area),"fraction_of_quota_triangle_area":float(2*current_area),
        "complexity_penalty":.02*leaf_count,"area_penalty":.005*float(current_area),"total_penalty":penalty,
        "reference_intersection_area":float(overlap),"reference_iou":float(overlap/(current_area+reference_area-overlap)),
        "area_outside_reference":float(current_area-overlap),"reference_area_not_covered":float(reference_area-overlap),
        "datasets":datasets}
    print(name, datasets["holdout"]["covered"], "/",len(queries),"area",float(current_area),flush=True)
