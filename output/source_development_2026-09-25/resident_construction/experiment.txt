def sha(data):
    return hashlib.sha256(data).hexdigest()

def save(path,value,*,compressed=False):
    path = OUT/path
    path.parent.mkdir(parents=True,exist_ok=True)
    raw=ir.json_bytes(value)
    path.write_bytes(lzma.compress(raw,preset=3) if compressed else raw)
    return {'path':str(path.relative_to(OUT)).replace('\\','/'),'sha256':sha(raw),'bytes':len(raw),
            'stored_sha256':sha(path.read_bytes()),'stored_bytes':path.stat().st_size}

def check(name,passed,**details):
    item={'name':name,'passed':bool(passed),**details}
    checks.append(item)
    save('checks.json',checks)
    print(json.dumps({'check':name,'passed':item['passed'],**({'details':details} if not passed else {})}),flush=True)
    return item['passed']

def compile_case(name,specification,*,graph=None):
    print('compiling '+name,flush=True)

    return {'name':name,'folder':folder,'definition':definition,'graph':actual_graph,'raw':raw,'manifest':manifest,

def run(label,backend,input_path,epochs,*,reverse=False,expected_exit=0):
    folder=OUT/label
    folder.mkdir(parents=True,exist_ok=False)
    output_path=folder/'checkpoint.bin'
    command=[str(EXE),'run','--backend',backend,'--input',str(input_path),'--output',str(output_path),'--epochs',str(epochs)]
    env=dict(os.environ)
    env.pop('DAWNWOOD_VALIDATION',None)
    env.pop('VK_LAYER_PATH',None)
    selected_env={}
    if backend=='vulkan':
        command.extend(['--device','RTX 5070 Ti'])
        selected_env={'DAWNWOOD_VALIDATION':'1','VK_LAYER_PATH':str(LAYER)}
        env.update(selected_env)
    if reverse:command.append('--reverse-targets')
    start=datetime.now(timezone.utc).isoformat()
    process=subprocess.run(command,env=env,capture_output=True)
    (folder/'stdout.json').write_bytes(process.stdout)
    (folder/'stderr.txt').write_bytes(process.stderr)
    receipt=json.loads(process.stdout)
    item={'label':label,'command':command,'environment_overrides':selected_env,'started_utc':start,
          'ended_utc':datetime.now(timezone.utc).isoformat(),'returncode':process.returncode,'expected_exit':expected_exit,
          'input_path':str(input_path.relative_to(OUT)).replace('\\','/'),'input_sha256':sha(input_path.read_bytes()),
          'output_path':str(output_path.relative_to(OUT)).replace('\\','/'),
          'output_sha256':sha(output_path.read_bytes()) if output_path.exists() else None,
          'executable_sha256':sha(EXE.read_bytes()),'shader_sha256':sha(SHADER.read_bytes()),'receipt':receipt}
    (folder/'execution.json').write_bytes(ir.json_bytes(item))
    return item,output_path

def collect(item):
    receipt,path=item
    receipts.append(receipt)
    save('execution_receipts.json',receipts)
    check(receipt['label']+'_exit',receipt['returncode']==receipt['expected_exit'],
          actual=receipt['returncode'],expected=receipt['expected_exit'])
    native=receipt['receipt']
    if native.get('backend')=='vulkan':
        check(receipt['label']+'_physical_gpu_validation',native.get('device_type')=='discrete_gpu'
              and 'RTX 5070 Ti' in native.get('device','') and native.get('validation_enabled') is True
              and native.get('validation_errors')==0 and native.get('validation_warnings')==0,
              device=native.get('device'),validation=native.get('validation_enabled'),
              errors=native.get('validation_errors'),warnings=native.get('validation_warnings'))
    print(json.dumps({'completed':receipt['label'],'execution_seconds':native.get('execution_seconds'),
                      'failed_lanes':native.get('failed_lanes')}),flush=True)
    return path

def compare(name,left,right):
    a,b=left.read_bytes(),right.read_bytes()
    changed=sum(x!=y for x,y in zip(struct.unpack('<'+'I'*(len(a)//4),a),struct.unpack('<'+'I'*(len(b)//4),b))) if len(a)==len(b) else None
    item={'name':name,'left':str(left.relative_to(OUT)).replace('\\','/'),
          'right':str(right.relative_to(OUT)).replace('\\','/'),'left_sha256':sha(a),'right_sha256':sha(b),
          'bytes_left':len(a),'bytes_right':len(b),'compared_words':len(a)//4 if len(a)==len(b) else None,
          'changed_words':changed,'all_bytes_equal':a==b}
    comparisons.append(item)
    save('comparisons.json',comparisons)
    check(name,a==b,changed_words=changed,bytes_left=len(a),bytes_right=len(b))

def pair(name,input_path,epochs,*,expected_exit=0):
    with ThreadPoolExecutor(max_workers=2) as pool:
        cpu=pool.submit(run,name+'_cpu','cpu',input_path,epochs,expected_exit=expected_exit)
        gpu=pool.submit(run,name+'_vulkan','vulkan',input_path,epochs,expected_exit=expected_exit)
        cpu_path=collect(cpu.result())
        gpu_path=collect(gpu.result())
    compare(name+'_cpu_gpu_exact',cpu_path,gpu_path)
    return cpu_path,gpu_path

def assess(name,path,manifest):
    report=results.evaluate_results(path.read_bytes(),manifest)
    save(name+'/independent_results.json.xz',report,compressed=True)
    rows=report['results']
    summary={'name':name,'checkpoint_sha256':report['checkpoint_sha256'],'count':report['count'],
        'all_assessments_passed':report['all_assessments_passed'],
        'all_runtime_healthy':report['all_runtime_healthy'],
        'all_programs_geometrically_admissible':report['all_programs_geometrically_admissible'],
        'program_node_counts':sorted(set(r['program']['node_count'] for r in rows if r['program'])),
        'coverage_values':sorted(set(r['native_diagnostics']['coverage'] for r in rows)),
        'admission_values':sorted(set(r['native_diagnostics']['admitted'] for r in rows)),
        'max_query_error':max((r['independent_readout']['query']['absolute_error'] for r in rows if r['independent_readout']),default=None),
        'max_witness_error':max((r['independent_readout']['witness']['absolute_error'] for r in rows if r['independent_readout']),default=None),
        'max_score_error_given_returned_coverage':max((abs(r['independent_readout']['objective']['score_signed_error_given_native_coverage']) for r in rows if r['independent_readout']),default=None),
        'coverage_ambiguous_rows':sum(bool(r['independent_readout']['objective']['ambiguous_training_indices']) for r in rows if r['independent_readout']),
        'unique_program_count':report['unique_program_count'],'unique_topology_count':report['unique_topology_count'],
        'unique_semantic_region_count':report['unique_semantic_region_count']}
    assessments.append(summary)
    save('assessment_summaries.json',assessments)
    check(name+'_independent_assessment',report['all_assessments_passed'])
    print(json.dumps({'assessment':summary}),flush=True)
    return report
