def run(name,backend,input_path,epochs,manifest):
    folder=OUT/name;folder.mkdir(exist_ok=False);output=folder/'checkpoint.bin'
    command=[str(EXE),'run','--backend',backend,'--input',str(input_path),'--output',str(output),'--epochs',str(epochs)]
    env=dict(os.environ);env.pop('DAWNWOOD_VALIDATION',None);env.pop('VK_LAYER_PATH',None);overrides={}
    if backend=='vulkan':
        command+=['--device','RTX 5070 Ti'];overrides={'DAWNWOOD_VALIDATION':'1','VK_LAYER_PATH':str(LAYER)};env.update(overrides)
    started=datetime.now(timezone.utc).isoformat();p=subprocess.run(command,env=env,capture_output=True)
    (folder/'stdout.json').write_bytes(p.stdout);(folder/'stderr.txt').write_bytes(p.stderr)
    receipt=json.loads(p.stdout)
    value={'name':name,'command':command,'environment_overrides':overrides,'started_utc':started,
        'finished_utc':datetime.now(timezone.utc).isoformat(),'returncode':p.returncode,'native':receipt,
        'input':str(input_path.relative_to(OUT)).replace('\\','/'),'input_sha256':sha(input_path.read_bytes()),
        'output':str(output.relative_to(OUT)).replace('\\','/'),'output_sha256':sha(output.read_bytes()),
        'executable_sha256':sha(EXE.read_bytes()),'shader_sha256':sha(SHADER.read_bytes())}
    save(Path(name)/'execution.json',value)
    return output,value
def collect(item):
    path,receipt=item;receipts.append(receipt);save('execution_receipts.json',receipts)
    check(receipt['name']+'_healthy',receipt['returncode']==0 and receipt['native']['failed_lanes']==0)
    if receipt['native']['backend']=='vulkan':
        x=receipt['native'];check(receipt['name']+'_physical_gpu_validation',x['device_type']=='discrete_gpu'
            and 'RTX 5070 Ti' in x['device'] and x['validation_enabled'] and x['validation_errors']==0 and x['validation_warnings']==0)
    return path
def compare(name,left,right):
    a,b=left.read_bytes(),right.read_bytes()
    changed=sum(x!=y for x,y in zip(struct.unpack('<'+'I'*(len(a)//4),a),struct.unpack('<'+'I'*(len(b)//4),b))) if len(a)==len(b) else None
    item={'name':name,'left_sha256':sha(a),'right_sha256':sha(b),'all_bytes_equal':a==b,
        'compared_words':len(a)//4 if len(a)==len(b) else None,'changed_words':changed}
    comparisons.append(item);save('comparisons.json',comparisons);check(name,a==b,changed_words=changed)
