def read(path):

    path = OUT / path
    path.parent.mkdir(parents=True, exist_ok=True)
    raw = ir.json_bytes(value)
    path.write_bytes(lzma.compress(raw, preset=3) if path.suffix == '.xz' else raw)

# Reduce the lane count for focused negative probes without altering retained
# lane contents or executable definitions. Native validation still runs fully.
def three(raw, manifest):
    decoded = resident.decode_checkpoint(raw)
    result = bytearray(raw[:decoded['instances_offset_bytes'] + 3*4*decoded['instance_words']])
    struct.pack_into('<I', result, 8, 3)
    result = bytes(result)
    copied = deepcopy(manifest)
    copied['count'] = 3
    copied['configuration_sha256'] = resident.decode_checkpoint(result)['configuration_sha256']
    copied['program_sha256'] = sha(result)
    return result, copied

def reject(name, raw, manifest, parent_raw=pr, parent_meta=pm, query=None):
    before = tuple(map(sha,(raw,parent_raw,ir.json_bytes(manifest),ir.json_bytes(parent_meta))))
    try:
        guidance.migrate_guidance(parent_raw,parent_meta,raw,manifest,query=query)
        item = {'name':name,'passed':False,'error':'accepted'}
    except ValueError as error:
        after = tuple(map(sha,(raw,parent_raw,ir.json_bytes(manifest),ir.json_bytes(parent_meta))))
        item = {'name':name,'passed':before==after,'inputs_unchanged':before==after,'error':str(error)}
    rejections.append(item)
    print(json.dumps(item),flush=True)
