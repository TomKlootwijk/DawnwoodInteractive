def write(name,value):
    data=(json.dumps(value,indent=2,ensure_ascii=False,allow_nan=False)+"\n").encode("utf-8")
    stored=lzma.compress(data) if name.endswith(".xz") else data
    (OUT/name).write_bytes(stored)
    return {"file_sha256":sha(stored),"uncompressed_json_sha256":sha(data),"bytes":len(stored)}

def load(path):
    data=path.read_bytes()
    if path.suffix==".xz":data=lzma.decompress(data)
    return json.loads(data.decode("utf-8"))

def rects(program,cert):
    result=[]
    for i in cert["leaf_indices"]:
        cx,cy,hx,hy=map(q,program["boxes"][i]);result.append((cx-hx,cx+hx,cy-hy,cy+hy))
    return result
def area(rectangles):return sum((r[1]-r[0])*(r[3]-r[2]) for r in rectangles)
def overlap(left,right):return sum(max(Fraction(),min(a[1],b[1])-max(a[0],b[0]))*max(Fraction(),min(a[3],b[3])-max(a[2],b[2])) for a in left for b in right)
