def save_program(name,definition,version=3):
    folder=OUT/name;folder.mkdir(exist_ok=False)
    backend=v3 if version==3 else v2
    raw,manifest=backend.compile_definition(model,definition)
    (folder/'program.bin').write_bytes(raw)
    for filename,obj in [('definition.json.xz',definition),('manifest.json.xz',manifest)]:
        (folder/filename).write_bytes(lzma.compress(ir.json_bytes(obj)))
    return folder/'program.bin',raw

def run(name,path,backend,epochs,version=3,expected=0,reverse=False):
    folder=OUT/name;folder.mkdir(exist_ok=False);output=folder/'checkpoint.bin'
    command=[str(EXES[version]),'run','--input',str(path),'--output',str(output),'--backend',backend,'--epochs',str(epochs)]
    if backend=='vulkan':command+=['--device','RTX 5070 Ti']
    if reverse:command+=['--reverse-targets']
    started=time.perf_counter();result=subprocess.run(command,capture_output=True,env=env)
    (folder/'stdout.json').write_bytes(result.stdout);(folder/'stderr.txt').write_bytes(result.stderr)
    receipt={'name':name,'command':command,'returncode':result.returncode,'expected_returncode':expected,
             'wall_seconds':time.perf_counter()-started,'input_sha256':sha(Path(path).read_bytes()),
             'output_sha256':sha(output.read_bytes()) if output.exists() else None}
    try:receipt['report']=json.loads(result.stdout)
    except Exception:receipt['report']=None
    receipts.append(receipt);(folder/'execution.json').write_bytes(ir.json_bytes(receipt));write('execution_receipts.json',receipts)
    assert result.returncode==expected,receipt
    if backend=='vulkan' and expected in (0,3):
        report=receipt['report'];assert report['validation_enabled'] and report['validation_errors']==report['validation_warnings']==0,receipt
        assert report['device_type']=='discrete_gpu' and '5070 Ti' in report['device'],receipt
    print(json.dumps({'name':name,'exit_code':result.returncode,'passed':True}),flush=True)
    if expected==2:return None,receipt
    raw=output.read_bytes();decoded=(v3 if version==3 else v2).decode_checkpoint(raw)
    return (raw,decoded,output),receipt

def check(name,condition,**details):
    row={'name':name,'passed':bool(condition),**details};checks.append(row);write('checks.json',checks)
    assert condition,row

def same_images(left,right):
    return left[0][left[1]['instances_offset_bytes']:]==right[0][right[1]['instances_offset_bytes']:]

    try:operation()
    except (ValueError,TypeError) as error:rejections.append({'name':name,'passed':True,'exception':type(error).__name__,'message':str(error)});return
    rejections.append({'name':name,'passed':False});raise AssertionError(name)
