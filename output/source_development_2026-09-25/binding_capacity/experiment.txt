def save(name, value):

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def check(name, passed, **details):
    checks.append(dict(name=name, passed=bool(passed), **details))

def run(name, rows, *, tag=None, expected_failure=False):
    entry = bank[name]
    compiled = ir.compile_binding(entry['binding'], name)
    assert max(x[0] for x in compiled['instructions']) <= 17, 'Old standalone XIR harness supports only0..17'
    label = tag or name
    folder = OUT / label
    folder.mkdir(exist_ok=True)
    rounded = [[ir.fp32(row[k],k)[0] for k in compiled['inputs']] for row in rows]
    raw = bytearray(b'DWIR0001')
    raw.extend(struct.pack('<4I',len(rows),len(compiled['instructions']),len(compiled['inputs']),len(compiled['output_names'])))
    for instruction in compiled['instructions']: raw.extend(struct.pack('<4I',*instruction))
    for register in compiled['output_registers']: raw.extend(struct.pack('<I',register))
    for row in rounded: raw.extend(struct.pack('<'+'f'*len(row),*row))
    input_path, output_path = folder/'input.bin', folder/'output.bin'
    input_path.write_bytes(raw)
    command = [str(EXE),'run','--backend','cpu','--input',str(input_path),'--output',str(output_path)]
    process = subprocess.run(command, capture_output=True, text=True)
    (folder/'stdout.txt').write_text(process.stdout,encoding='utf-8')
    (folder/'stderr.txt').write_text(process.stderr,encoding='utf-8')
    result = ir.read_result(output_path)
    named = [dict(zip(compiled['output_names'],row)) for row in result['outputs']]
    save(f'{label}/receipt.json', dict(binding=name,command=command,returncode=process.returncode,
        input_names=compiled['inputs'],output_names=compiled['output_names'],input_rows=rounded,
        input_sha256=hashlib.sha256(raw).hexdigest(),result=result))
    calls.append(dict(label=label,binding=name,rows=len(rows),returncode=process.returncode,
                      failed_rows=sum(s!=0 for s in result['statuses'])))
    check(label+'_native_status', process.returncode==(3 if expected_failure else 0)
          and result['all_status_ok']==(not expected_failure), statuses=result['statuses'])
    return named, result
