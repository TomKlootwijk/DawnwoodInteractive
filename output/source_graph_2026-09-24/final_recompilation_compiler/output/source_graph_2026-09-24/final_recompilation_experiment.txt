def mapping_audit(definition):
    source=definition['source'];graph=source['application_graph'];checked=[]
    def check(mapping):
        plan_name=mapping['plan'];plan=definition[plan_name+'_plan']
        provenance=next(n for n in graph[plan_name]['nodes'] if n['node']==mapping['node'])
        assert mapping['source_pointer']==provenance['source_pointer']
        assert mapping['plan_step_start']==provenance['plan_step_start']
        assert mapping['plan_step_end_exclusive']==provenance['plan_step_end_exclusive']
        assert mapping['output_registers']==provenance['output_registers']
        step=plan[mapping['call_step']]
        assert mapping['plan_step_start']<=mapping['call_step']<mapping['plan_step_end_exclusive']<=len(plan)
        assert step[0] in (3,4,11)
        input_base,output_base=(step[2],step[3]) if step[0]==3 else (step[4],step[5])
        assert list(mapping['argument_registers'].values())==list(range(input_base,input_base+len(mapping['argument_registers'])))
        assert list(mapping['output_registers'].values())==list(range(output_base,output_base+len(mapping['output_registers'])))
        checked.append({'plan':plan_name,'node':mapping['node'],'call_step':mapping['call_step']})
    def retention(policy):
        assert not set(policy).intersection({'step','inputs','argument_base','output_base'})
        assert policy['layout_source']==gp.graph_compiler.PROFILE and policy['graph_calls']
        for mapping in policy['graph_calls']:check(mapping)
    for metadata in (source.get('source_slot_bindings'),source.get('authoring',{}).get('slot_resolutions')):
        if metadata:
            for policy in metadata['mutation_program_retention']:retention(policy)
    for metadata in (source.get('catalogue_extension'),source.get('authoring',{}).get('catalogue_extension')):
        if metadata and metadata.get('added'):
            retention(metadata['mutation_retention'])
            routing=metadata['routing']
            assert 'call_step' not in routing and routing['layout_source']==gp.graph_compiler.PROFILE and routing['graph_calls']
            for mapping in routing['graph_calls']:check(mapping)
    returned=[]
    if source.get('enzyme'):
        application=source['enzyme']['resident_application']
        assert not set(application).intersection({'action_step_start','action_step_end_exclusive','original_action_prefix_preserved','base_action_steps'})
        assert application['layout_source']==gp.graph_compiler.PROFILE
        expected=graph['structural_fidelity']['enzyme']
        assert {role:value['node'] for role,value in application['graph_calls'].items()}==expected
        for mapping in application['graph_calls'].values():check(mapping)
        for item in application['return_writes']:
            step=definition['action_plan'][item['step']]
            assert step[0]==6 and definition['state_names'][step[1]]==item['field']
            returned.append({'field':item['field'],'step':item['step']})
        assert set(x['field'] for x in returned)==set(source['enzyme']['state_layout'])
    authoring=source.get('authoring',{})
    assert source.get('source_slot_bindings')==authoring.get('slot_resolutions')
    assert source.get('catalogue_extension')==authoring.get('catalogue_extension')
    for scope in ('mutation','action'):
        assert graph[scope]['plan_sha256']==semantic(definition[scope+'_plan'])
    return {'passed':True,'call_mappings_checked':checked,'enzyme_return_writes_checked':returned,
            'duplicated_authoring_policies_equal':True,'both_current_plan_hashes_match':True}

def compact_manifest(manifest):
    return {key:manifest.get(key) for key in ('scope','status','complete_application_lowered','source_cycle_lowered',
            'source_graph_lowered','source_graph_profile','domain_application_implemented','program_sha256','configuration_sha256')}

def manifest_audit(manifest,definition):
    assert manifest['complete_application_lowered'] is True
    assert manifest['source_cycle_lowered'] is True and manifest['source_graph_lowered'] is True
    assert manifest['source_graph_profile']==gp.graph_compiler.PROFILE
    assert manifest['status']=='explicit_source_graph_lowered_numerical_evidence_recorded_separately'
    assert manifest['domain_application_implemented']==('enzyme' in definition['source'])
    assert manifest['binding_contract']['application_graph']==definition['source']['application_graph']
    assert manifest['binding_contract']['source_authoring']==definition['source']['authoring']
    return {'passed':True,**compact_manifest(manifest)}

def archive_files(folder):
    names=('definition.json.xz','manifest.json.xz','program.bin','source_model.json','source_graph.json','enzyme_inputs.json')
    return {name:{'sha256':sha((folder/name).read_bytes()),'bytes':(folder/name).stat().st_size}
            for name in names if (folder/name).exists()}

def compare_program(raw,expected):
    result={'complete_program_bytes_equal':raw==expected,'current_program_sha256':sha(raw),
            'archived_program_sha256':sha(expected),'current_bytes':len(raw),'archived_bytes':len(expected)}
    if raw!=expected:
        result['first_different_byte']=next((i for i,(a,b) in enumerate(zip(raw,expected)) if a!=b),min(len(raw),len(expected)))
    return result

def rebuild(folder, *, model=None, graph=None, override_instances=False, classification='source_graph_application'):
    archived=load(folder/'definition.json.xz')
    model=load(folder/'source_model.json') if model is None else model
    graph=load(folder/'source_graph.json') if graph is None else graph
    cycle=deepcopy(archived['source']['cycle_definition'])
    kwargs={};enzyme=archived.get('source',{}).get('enzyme')
    if enzyme:
        kwargs['enzyme_inputs']=deepcopy(enzyme.get('input_document') or load(OUT/'enzyme/enzyme_inputs.json'))
    current=gp.build_definition(model,cycle,graph,len(archived['instances']),**kwargs)
    original_initializers_equal=current['instances']==archived['instances']
    if override_instances:current['instances']=deepcopy(archived['instances'])
    raw,manifest=gp.compile_definition(model,current)
    layout=mapping_audit(current);flags=manifest_audit(manifest,current)
    row={'case':relative(folder),'classification':classification,'count':len(current['instances']),
         'input_files':archive_files(folder),'source_model_semantic_sha256':semantic(model),
         'source_graph_semantic_sha256':semantic(graph),'source_cycle_semantic_sha256':semantic(cycle),
         'enzyme_inputs_semantic_sha256':semantic(kwargs['enzyme_inputs']) if enzyme else None,
         'instance_policy':'Archived explicit initial instances restored after rebuilding the source graph.' if override_instances else 'Current builder with inferred archived count and exact retained enzyme observations.',
         'builder_initializers_equal_archive_before_override':original_initializers_equal,
         'current_definition_semantic_sha256':semantic(current),'current_manifest_semantic_sha256':semantic(manifest),
         'live_layout_audit':layout,'current_manifest':flags,
         'historical_manifest':compact_manifest(load(folder/'manifest.json.xz')),
         **compare_program(raw,(folder/'program.bin').read_bytes())}
    row['passed']=row['complete_program_bytes_equal'] and layout['passed'] and flags['passed']
    print(json.dumps({'case':folder.name,'passed':row['passed'],'count':row['count'],'classification':classification}),flush=True)
    return row,current,model,graph

def capture(node,which,names,prefix):
    call=calls[node['id']]
    if which=='input':
        ordered=list(node['args']);base=call['inputs'][0];at=call['step']
    else:
        if node['op']=='call_helper':function=application['functions'][node['function']]
        else:
            key=node.get('contract',{}).get('key',node.get('record',{}).get('key'))
            record=next(item for item in application['records'] if item['key']==key)
            name=application['families'][record['body']]['methods'][str(node['role'])]
            function=application['functions'][name]
        ordered=list(function['binding']['outputs']);base=call['output_base'];at=call['step']+1
    for name in names:captures.append({'name':prefix+name,'at':at,'register':base+ordered.index(name)})
