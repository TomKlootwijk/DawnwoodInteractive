def verify_layout(d):
    source=d['source']
    mapped=0
    def call(item):
        nonlocal mapped
        plan=d[item['plan']+'_plan']; step=plan[item['call_step']]
        assert item['plan_step_start']<=item['call_step']<item['plan_step_end_exclusive']<=len(plan)
        assert step[0] in (3,4,11)
        argument_base,output_base=(step[2],step[3]) if step[0]==3 else (step[4],step[5])
        assert list(item['argument_registers'].values())==list(range(argument_base,argument_base+len(item['argument_registers'])))
        assert list(item['output_registers'].values())==list(range(output_base,output_base+len(item['output_registers'])))
        mapped+=1
    def policy(item):
        assert not set(item).intersection({'step','inputs','argument_base','output_base'})
        assert item['layout_source']==compiler.PROFILE and item['graph_calls']
        for mapping in item['graph_calls']: call(mapping)
    for metadata in (source.get('source_slot_bindings'),source.get('authoring',{}).get('slot_resolutions')):
        if metadata:
            for item in metadata['mutation_program_retention']: policy(item)
    for metadata in (source.get('catalogue_extension'),source.get('authoring',{}).get('catalogue_extension')):
        if metadata and metadata.get('added'):
            policy(metadata['mutation_retention'])
            routing=metadata['routing']
            assert 'call_step' not in routing and routing['layout_source']==compiler.PROFILE and routing['graph_calls']
            for item in routing['graph_calls']: call(item)
    if source.get('enzyme'):
        application=source['enzyme']['resident_application']
        assert not set(application).intersection({'action_step_start','action_step_end_exclusive','original_action_prefix_preserved','base_action_steps'})
        assert application['layout_source']==compiler.PROFILE
        for item in application['graph_calls'].values(): call(item)
        for item in application['return_writes']:
            step=d['action_plan'][item['step']]
            assert step[0]==6 and d['state_names'][step[1]]==item['field']
    authoring=source.get('authoring',{})
    assert source.get('source_slot_bindings')==authoring.get('slot_resolutions')
    assert source.get('catalogue_extension')==authoring.get('catalogue_extension')
    return {'passed':True,'mapped_calls_checked':mapped,'duplicated_policies_match':True}

def nodes(g):
    return [n for phase in g['phases'] for n in phase['nodes']]

def first_call(g):
    return next(n for n in nodes(g) if n['op'] == 'call_record')

def helper(width=1, inputs=0):
    ins = [f'x{i}' for i in range(inputs)]
    outputs = {f'y{i}': i + 1 for i in range(width)}
    return {'binding': {'inputs': ins, 'outputs': outputs},
            'inputs': dict.fromkeys(ins, 'coefficient'),
            'outputs': dict.fromkeys(outputs, 'coefficient')}

def probe(name, edit, *, expected='reject', base=core, source_graph=graph, source_model=model):
    d, g = deepcopy(base), deepcopy(source_graph)
    before = digest(d)
    row = {'name': name, 'expected': expected}
    try:
        edit(g, d)
        # Atomicity is relative to the exact input passed into apply_graph.
        before = digest(d)
        metadata = compiler.apply_graph(d, cycle, g)
        layout=verify_layout(d)
        program, manifest = wrapper.compile_definition(source_model, d)
        row.update(outcome='accepted', program_sha256=ir.sha256(program),
                   mutation_peak=metadata['mutation']['value_register_high_water'],
                   action_peak=metadata['action']['value_register_high_water'],
                   mutation_steps=len(d['mutation_plan']), action_steps=len(d['action_plan']),
                   family_mapping_unchanged=d['families'] == base['families'],
                   records_unchanged=d['records'] == base['records'],live_layout_metadata=layout,
                   complete_application_lowered=manifest['complete_application_lowered'])
    except Exception as error:
        row.update(outcome='rejected', exception=type(error).__name__, message=str(error),
                   definition_unchanged=digest(d) == before)
    row['passed'] = ((expected == 'reject' and row['outcome'] == 'rejected' and row['definition_unchanged'])
                     or (expected == 'accept' and row['outcome'] == 'accepted'))
    results.append(row)
    print(json.dumps({'name': name, 'passed': row['passed'], 'outcome': row['outcome']}))
    return row

def add_helper(g, entry=None):
    g['functions']['experiment_helper'] = deepcopy(entry or helper())
    return g['functions']['experiment_helper']

def call_body_as_helper(g,d):
    node=next(n for n in nodes(g) if n['op']=='call_record' and n['slot']=='body' and 'key' in n['record'])
    rec=next(r for r in d['records'] if r['key']==node['record']['key'])
    fn=d['families'][rec['body']]['methods'][str(node['role'])]
    identity,args=node['id'],node['args']
    node.clear(); node.update(id=identity,op='call_helper',function=fn,args=args)
probe('helper_cannot_bypass_resident_body_dispatch',call_body_as_helper)

def unused_helper(g,d):

def liveness(g,d,count):
    g['functions']['produce32']=helper(width=32)
    g['functions']['consume64']=helper(inputs=64)
    target=g['phases'][-1]['nodes']
    for i in range(count): target.append({'id':f'hold_{i}','op':'call_helper','function':'produce32','args':{}})
    for i in range(0,count,2):
        refs=[{'node':f'hold_{j}','output':f'y{k}'} for j in range(i,min(i+2,count)) for k in range(32)]
        while len(refs)<64: refs.append({'literal':0,'type':'coefficient'})
        target.append({'id':f'consume_{i}','op':'call_helper','function':'consume64','args':dict(zip((f'x{k}' for k in range(64)),refs))})
probe('liveness_large_accepted',lambda g,d:liveness(g,d,4),expected='accept')
probe('liveness_overflow_atomic',lambda g,d:liveness(g,d,7))
