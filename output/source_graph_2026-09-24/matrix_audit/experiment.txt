def save_json(name,value):
    (OUT/name).write_bytes(c.ir.json_bytes(value))

def save_definition(name,d):
    folder=OUT/name;folder.mkdir(exist_ok=False)
    raw,manifest=r.compile_definition(MODEL,d)
    (folder/'program.bin').write_bytes(raw)
    for fname,obj in [('definition.json',d),('manifest.json',manifest)]:
        data=c.ir.json_bytes(obj);(folder/(fname+'.xz')).write_bytes(lzma.compress(data))
    return folder/'program.bin',manifest

def run_pair(name,source,epochs,expected=0):
    folder=OUT/name;folder.mkdir(exist_ok=False)
    reports={};outputs={}
    for backend in ['cpu','vulkan']:
        output=folder/(backend+'.bin')
        command=[str(EXE),'run','--input',str(source),'--output',str(output),'--backend',backend,'--epochs',str(epochs)]
        if backend=='vulkan':command+=['--device','RTX 5070 Ti']
        started=time.perf_counter();run=subprocess.run(command,capture_output=True,env=ENV)
        elapsed=time.perf_counter()-started
        (folder/(backend+'.stdout.log')).write_bytes(run.stdout)
        (folder/(backend+'.stderr.log')).write_bytes(run.stderr)
        report=json.loads(run.stdout.decode('utf-8'));reports[backend]=report
        receipt={'command':command,'returncode':run.returncode,'wall_seconds':elapsed,'report':report,
                 'input_sha256':c.ir.sha256(Path(source).read_bytes()),'output_sha256':c.ir.sha256(output.read_bytes()) if output.exists() else None}
        (folder/(backend+'.execution.json')).write_bytes(c.ir.json_bytes(receipt))
        if run.returncode!=(0 if expected==0 else 3):raise AssertionError((name,backend,receipt))
        outputs[backend]=output.read_bytes()
    assert outputs['cpu']==outputs['vulkan'],name
    decoded=r.decode_checkpoint(outputs['cpu'])
    assert all(i['status']==expected for i in decoded['instances']),name
    vr=reports['vulkan']
    assert vr['validation_enabled'] and vr['validation_errors']==0 and vr['validation_warnings']==0,vr
    row={'name':name,'epochs':epochs,'count':decoded['count'],'status':expected,
         'complete_checkpoints_bit_identical':True,'validation_enabled':True,'validation_errors':0,'validation_warnings':0,
         'compared_instance_words':decoded['count']*decoded['instance_words']}
    receipts.append(row);save_json('paired_summary.json',receipts)
    print(json.dumps(row),flush=True)
    return decoded,outputs['cpu']

def inputs(d):
    # Preserve source seed wavefront. Vary the chart, neck, and external phase.
    for i,instance in enumerate(d['instances']):
        s=instance['state'];s['surface_u']=(i%17)/17;s['surface_v']=(i//17)/17
        s['neck']=i%2;s['phase']=((i%13)-6)/8
        # The original selected six-primitive route is adjusted for neck.
        for bit,value in enumerate([0,1,0,0]):s[f'route_b{bit}']=value^(i%2)
    return d

def scaled(actual,reference):
    return math.sqrt(sum((a-b)**2 for a,b in zip(actual,reference)))/max(1,math.sqrt(sum(x*x for x in reference)))
def mat_apply(A,w):
    return [A[0]*w[0]+A[1]*w[2], A[0]*w[1]+A[1]*w[3],
            A[2]*w[0]+A[3]*w[2], A[2]*w[1]+A[3]*w[3]]

    off=.02*math.sin(v['T_in_crystal_ar'])
    expected_T=[1+.1*math.hypot(w[0],w[1])+.01*abs(v['T_in_field_distance']),off,off,
                1+.1*math.hypot(w[2],w[3])+.01*abs(v['T_in_field_distance'])]
    T=[v[n] for n in ['t00','t01','t10','t11']];A=[v[n] for n in ['a00','a01','a10','a11']]
    det=T[0]*T[3]-T[1]*T[2];expected_A=[T[3]/det,-T[1]/det,-T[2]/det,T[0]/det]
    residual=max(abs(T[0]*A[0]+T[1]*A[2]-1),abs(T[0]*A[1]+T[1]*A[3]),
                 abs(T[2]*A[0]+T[3]*A[2]),abs(T[2]*A[1]+T[3]*A[3]-1))
    raw=[v['raw_'+n] for n in c.PAIR];actual_conditioned=[v['conditioned_'+n] for n in c.PAIR]
