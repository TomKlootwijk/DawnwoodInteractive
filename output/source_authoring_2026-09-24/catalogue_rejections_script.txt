def reject(name,operation,description):
    result={'name':name,'description':description,'expected':'reject'}
    try:operation()
    except (ValueError,TypeError,KeyError,OverflowError) as error:
        result.update(rejected=True,error_type=type(error).__name__,message=str(error))
    except Exception as error:
        result.update(rejected=False,unexpected_exception=type(error).__name__,message=str(error))
    else:result.update(rejected=False,message='Unexpected acceptance')
    cases.append(result)
