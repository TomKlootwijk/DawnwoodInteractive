def digest(value):
    return hashlib.sha256(base.ir.json_bytes(value)).hexdigest()


def record(definition, key):
    return next(row for row in definition["records"] if row["key"] == key)


def slot_binding(definition, key, slot, role="0"):
    r = record(definition, key)
    if slot != "body":
        fn = deepcopy(definition["functions"][r[slot]]["binding"])
        name = next(iter(fn["outputs"]))
        fn["outputs"][name] = {"op": "add", "args": [fn["outputs"][name], .0001]}
        return {"kind": slot, "function": fn, "mutation_policy": "preserve_program"}
    interface = definition["families"][r["body"]]["interface"]
    variants = {}
    for name, family in definition["families"].items():
        if family["interface"] != interface:
            continue
        fn = deepcopy(definition["functions"][family["methods"][role]]["binding"])
        output = "control" if key == "mutation" else next(iter(fn["outputs"]))
        fn["outputs"][output] = {"op": "add", "args": [fn["outputs"][output], .0001]}
        variants[name] = {"inherit": True, "roles": {role: fn}}
    return {"kind": "body", "variants": variants, "mutation_policy": "declared_variants"}


def edited(key="circle", slot="body", definition=core, role="0"):
    source = deepcopy(model)
    binding = slot_binding(definition, key, slot, role)
    source["operators"][base.INDEX[key]][slot] = {"resident_binding": binding}
    return source, binding


def execute(source, registry=None, chemical_profile=False):
    canonical, resolutions = slots.prepare_model(source, registry)
    definition = (enzyme.build_definition(canonical, cycle, inputs) if chemical_profile
                  else base.build_definition(canonical, cycle, 1))
    before = digest(definition)
    try:
        provenance = slots.apply_resolutions(source, definition, resolutions)
    except Exception as error:
        error.definition_unchanged = before == digest(definition)
        raise
    program, manifest = resident.compile_definition(source, definition)
    return {"program_sha256": hashlib.sha256(program).hexdigest(),
            "functions": len(definition["functions"]), "families": len(definition["families"]),
            "mutation_steps": len(definition["mutation_plan"]),
            "resolutions": len(provenance["resolutions"]),
            "retention": provenance["mutation_program_retention"]}


def probe(name, expected, function):
    row = {"case": name, "expected": expected}
    try:
        row["result"] = function()
        row["observed"] = "accepted"
    except (ValueError, TypeError, KeyError, RecursionError) as error:
        row.update(observed="rejected", exception=type(error).__name__, message=str(error))
        if hasattr(error, "definition_unchanged"):
            row["definition_unchanged"] = error.definition_unchanged
    except Exception as error:
        row.update(observed="unexpected_exception", exception=type(error).__name__,
                   message=str(error), traceback=traceback.format_exc())
    row["passed"] = row["observed"] == expected and row.get("definition_unchanged", True)
    rows.append(row)


def default_bytes(definition, chemical_profile=False):
    initial, _ = resident.compile_definition(model, deepcopy(definition))
    result = execute(model, chemical_profile=chemical_profile)
    result["original_program_sha256"] = hashlib.sha256(initial).hexdigest()
    result["identical"] = result["program_sha256"] == result["original_program_sha256"]
    if not result["identical"]:
        raise RuntimeError("Default program bytes changed")
    return result

def bypass_unresolved():
    source = deepcopy(model); source["operators"][9]["body"] = {"symbol": "not_resolved"}
    definition = deepcopy(core); before = digest(definition)
    try:
        slots.apply_resolutions(source, definition, [])
    except Exception as error:
        error.definition_unchanged = before == digest(definition)
        raise


probe("direct_apply_cannot_bypass_resolution", "rejected", bypass_unresolved)


def tamper_inline():
    source, _ = edited()
    canonical, resolutions = slots.prepare_model(source)
    resolutions[0]["binding"]["variants"]["circle"]["roles"]["0"]["outputs"]["value"] = 0
    definition = base.build_definition(canonical, cycle, 1); before = digest(definition)
    try:
        slots.apply_resolutions(source, definition, resolutions)
    except Exception as error:
        error.definition_unchanged = before == digest(definition)
        raise


probe("inline_resolution_cannot_diverge_from_source", "rejected", tamper_inline)
