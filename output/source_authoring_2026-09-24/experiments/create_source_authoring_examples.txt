def binding_metadata(binding,meaning):
    binding['source']='DWI-SOURCE-PROGRAM-0.1 explicit example model declaration'
    binding['meaning']=meaning
    binding['status']='Authored numerical definition; validation is recorded separately.'
    return binding

def replace_input(value,name,expression):
    if value=={'input':name}:return deepcopy(expression)
    if isinstance(value,dict):return {k:replace_input(v,name,expression) for k,v in value.items()}
    if isinstance(value,list):return [replace_input(v,name,expression) for v in value]
    return value

edited=deepcopy(model);variants={}
for family in ['hadamard','hadamard_conjugate']:
    role=deepcopy(baseline['functions'][baseline['families'][family]['methods']['1']]['binding'])
    role['outputs']=replace_input(role['outputs'],'phase',c.add(c.V('phase'),.125))
    binding_metadata(role,'Same typed log-polar double-Hadamard formula with an explicitly authored +0.125 radian phase offset; existing complex role0 is inherited.')
    variants[family]={'inherit':True,'roles':{'1':role}}
edited['operators'][1]['body']={'resident_binding':{'kind':'body','mutation_policy':'declared_variants','variants':variants}}
field=deepcopy(baseline['functions']['disk']['binding'])
field['outputs']={'distance':c.sub(c.norm2(c.sub(c.V('dx'),.025),c.add(c.V('dy'),.03125)),c.V('radius'))}

edited['operators'][1]['placement']={'resident_binding':{'kind':'placement','function':placement,'mutation_policy':'preserve_program'}}
(root/'authored_hadamard_model.json').write_bytes(c.ir.json_bytes(edited))
