from pathlib import Path
from copy import deepcopy
import json,math,random,sys
sys.path.insert(0,str(Path.cwd()));sys.path.insert(0,str(Path.cwd()/'tmp'))
from local_lab import source_cycle as c,source_enzyme as e,source_program as p,source_resident_v2 as r
from run_resident_enzyme_campaign import paired

BASE=Path('output/source_authoring_2026-09-24').resolve();BASE.mkdir(exist_ok=True)
MODEL,_=c.ir.read_json(c.MODEL);CYCLE,_=c.ir.read_json(c.CYCLE)
ROWS=[];COMPARISONS=[];CHECKPOINTS={}

def save(name,model,definition):
    out=BASE/name;out.mkdir(exist_ok=False)
    program,manifest=r.compile_definition(model,definition)
    (out/'source_model.json').write_bytes(c.ir.json_bytes(model))
    (out/'definition.json').write_bytes(c.ir.json_bytes(definition))
    (out/'program.bin').write_bytes(program)
    (out/'manifest.json').write_bytes(c.ir.json_bytes(manifest))
    return out/'program.bin'

def run(name,source,epochs,reverse=False):
    row,decoded,raw=paired(BASE/'runs',name,source,epochs,reverse)
    ROWS.append(row);CHECKPOINTS[name]=(decoded,raw)
    (BASE/'paired_summary.json').write_bytes(c.ir.json_bytes(ROWS))
    return decoded,raw

def compare(name,left,right,indices=None):
    a=CHECKPOINTS[left][0];b=CHECKPOINTS[right][0]
    indices=range(a['state_width']) if indices is None else indices
    counts=[sum(x['state_bits'][i]!=y['state_bits'][i] for i in indices) for x,y in zip(a['instances'],b['instances'])]
    result={'name':name,'left':left,'right':right,'state_indices':list(indices),'changed_instances':sum(bool(x) for x in counts),'changed_words':sum(counts)}
    COMPARISONS.append(result);return result

def main():
    base=c.build_definition(MODEL,CYCLE,129)
    authored=p.build_definition(MODEL,CYCLE,129)
    old,_=r.compile_definition(MODEL,base);new,_=r.compile_definition(MODEL,authored)
    assert old==new
    source=save('baseline',MODEL,authored)
    for ep in [1,2,64]:run(f'baseline_{ep}',source,ep)
    edited,_=c.ir.read_json(Path('source_bindings/examples/authored_hadamard_model.json'))
    for variant,slots in [('body',['body']),('field',['field']),('placement',['placement']),('combined',['body','field','placement'])]:
        model=deepcopy(MODEL)
        for slot in slots:model['operators'][1][slot]=deepcopy(edited['operators'][1][slot])
        definition=p.build_definition(model,CYCLE,129);src=save('hadamard_'+variant,model,definition)
        for ep in ([1,2,64] if variant=='combined' else [1,64]):
            name=f'{variant}_{ep}';dec,_=run(name,src,ep)
            result=compare(f'{variant}_causality_{ep}',name,f'baseline_{ep}',[3,4,5,6]);assert result['changed_instances']>0
            if variant in ['field','placement','combined']:
                for slot in [s for s in slots if s!='body']:
                    handle=list(definition['functions']).index(definition['records'][1][slot]);word=1 if slot=='field' else 2
                    assert all(i['records'][1]['raw_words'][word]==handle for i in dec['instances'])
            assert all(i['records'][1]['generation']==ep for i in dec['instances'])
        if variant=='combined':
            dec,raw=run('combined_resume_1_plus_63',BASE/'runs/combined_1/cpu.bin',63)
            assert raw==CHECKPOINTS['combined_64'][1]
            dec,raw=run('combined_reverse_64',src,64,True);assert raw==CHECKPOINTS['combined_64'][1]
    # A source-owned proposal edit retains the complete chemical law bank.
    inputs,_=c.ir.read_json(e.DEFAULT_INPUTS)
    chemical,_=c.ir.read_json(Path('source_bindings/examples/authored_enzyme_model.json'))
    edef=e.build_definition(MODEL,CYCLE,inputs);ce=p.build_definition(chemical,CYCLE,enzyme_inputs=inputs)
    for name in e.PROTECTED_ROLES.values():assert edef['functions'][name]==ce['functions'][name]
    for label,model,definition in [('enzyme_original',MODEL,edef),('enzyme_authored',chemical,ce)]:
        src=save(label,model,definition)
        for ep in [1,8,128]:run(label+'_'+str(ep),src,ep)
    for ep in [1,8,128]:compare('enzyme_proposal_edit_'+str(ep),'enzyme_authored_'+str(ep),'enzyme_original_'+str(ep),list(range(44,64)))
    assert COMPARISONS[-3]['changed_instances']>0
    # Added source record executes through its literal five-bit implicit index.
    model,_=c.ir.read_json(Path('source_bindings/examples/amplitude_budget_model.json'))
    definition=p.build_definition(model,CYCLE,129);src=save('amplitude_budget',model,definition)
    for ep in [1,2,64,2048]:
        dec,_=run('amplitude_'+str(ep),src,ep)
        assert all(i['state'][41]==31 for i in dec['instances'])
        assert all(i['records'][31]['generation']==ep for i in dec['instances'])
        rec=definition['records'][31]
        for slot,word in [('field',1),('placement',2)]:
            handle=list(definition['functions']).index(rec[slot])
            assert all(i['records'][31]['raw_words'][word]==handle for i in dec['instances'])
    dec,raw=run('amplitude_resume_1_plus_63',BASE/'runs/amplitude_1/cpu.bin',63);assert raw==CHECKPOINTS['amplitude_64'][1]
    dec,raw=run('amplitude_reverse_64',src,64,True);assert raw==CHECKPOINTS['amplitude_64'][1]
    # Different fifth-bit and far-end addresses use actual source indices.
    for index in [32,62]:
        changed=deepcopy(model);changed['operators'][31]['index']=index
        d=p.build_definition(changed,CYCLE,17)
        for lane in d['instances']:lane['state']['neck']=1
        p.catalogue.set_routes(d,[{'index':index}])
        source=save('route_'+str(index),changed,d)
        dec,_=run('route_'+str(index),source,2)
        assert all(i['state'][41]==index for i in dec['instances'])
    # Both matching symbolic-registry resolution and inline authoring execute identically.
    registered=deepcopy(MODEL);decl={'symbol':'User_authored_double_H_phase'}
    registered['operators'][1]['body']=decl
    reg={'profile':'DWI-SOURCE-SLOTS-0.1','entries':[{'key':'hadamard','slot':'body','declaration':decl,'binding':edited['operators'][1]['body']['resident_binding']}]}
    d=p.build_definition(registered,CYCLE,129,registry=reg);program,manifest=r.compile_definition(registered,d)
    assert program==(BASE/'hadamard_body/program.bin').read_bytes()
    (BASE/'registry.json').write_bytes(c.ir.json_bytes(reg))
    (BASE/'registry_model.json').write_bytes(c.ir.json_bytes(registered))
    (BASE/'comparisons.json').write_bytes(c.ir.json_bytes(COMPARISONS))
    summary={'default_binary_identical':True,'registry_and_inline_binary_identical':True,'paired_runs':len(ROWS),
        'compared_instance_words':sum(x['compared_instance_words'] for x in ROWS),
        'instance_epoch_steps_per_backend':sum(x['epochs']*x['count'] for x in ROWS),
        'slot_programs_retained_after_mutation':True,'old_snapshot_reverse_order_exact':True,'checkpoint_resume_exact':True,
        'protected_enzyme_law_functions_identical':True,'actual_added_source_indices_reached':[31,32,62],
        'all_checkpoint_bytes_identical':True,'validation_errors':0,'validation_warnings':0}
    (BASE/'summary.json').write_bytes(c.ir.json_bytes(summary));print(json.dumps(summary),flush=True)

if __name__=='__main__':main()
