cmake_minimum_required(VERSION 3.24)
project(DawnwoodInteractive VERSION 0.4.0 LANGUAGES CXX)
option(DW_ENABLE_CUDA "Build the native CUDA texture kernel" ON)
option(DW_SANITIZE_CPU "Enable address/undefined-behaviour sanitizers for CPU tests" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(dawnwood_core INTERFACE)
target_include_directories(dawnwood_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(dawnwood_core_tests tests/core_tests.cpp)
target_link_libraries(dawnwood_core_tests PRIVATE dawnwood_core)
add_executable(dawnwood_cpu src/cpu_main.cpp)
target_link_libraries(dawnwood_cpu PRIVATE dawnwood_core)
foreach(target dawnwood_core_tests dawnwood_cpu)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /fp:strict)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
    if(DW_SANITIZE_CPU)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endforeach()
enable_testing()
add_executable(dawnwood_integer_math_tests tests/integer_math_tests.cpp)
target_link_libraries(dawnwood_integer_math_tests PRIVATE dawnwood_core)
add_executable(dawnwood_integer_core_tests tests/integer_core_tests.cpp)
target_link_libraries(dawnwood_integer_core_tests PRIVATE dawnwood_core)
add_test(NAME dawnwood_integer_math COMMAND dawnwood_integer_math_tests)
add_test(NAME dawnwood_integer_core COMMAND dawnwood_integer_core_tests)
add_executable(dawnwood_integer_pair_cache_tests tests/integer_pair_cache_tests.cpp)
target_link_libraries(dawnwood_integer_pair_cache_tests PRIVATE dawnwood_core)
add_test(NAME dawnwood_integer_pair_cache COMMAND dawnwood_integer_pair_cache_tests)
add_executable(dawnwood_integer_predicate_tests tests/integer_predicate_tests.cpp)
target_link_libraries(dawnwood_integer_predicate_tests PRIVATE dawnwood_core)
add_test(NAME dawnwood_integer_predicates COMMAND dawnwood_integer_predicate_tests)
add_test(NAME dawnwood_core COMMAND dawnwood_core_tests)
add_test(NAME dawnwood_cpu_bc5 COMMAND dawnwood_cpu --steps 64)
add_test(NAME dawnwood_cpu_rg8 COMMAND dawnwood_cpu --exact --steps 64)
if(DW_ENABLE_CUDA)
  include(CheckLanguage)
  check_language(CUDA)
  if(NOT CMAKE_CUDA_COMPILER)
    message(FATAL_ERROR "CUDA compiler not found. Install CUDA >=12.8; use -DDW_ENABLE_CUDA=OFF for CPU tests only.")
  endif()
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES "120-real;120-virtual")
  endif()
  enable_language(CUDA)
  if(CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 12.8)
    message(FATAL_ERROR "This RTX 5070 Ti target needs nvcc >=12.8 for sm_120.")
  endif()
  find_package(CUDAToolkit 12.8 REQUIRED)
  foreach(target dawnwood dawnwood_reference)
    add_executable(${target} src/main.cu)
    target_link_libraries(${target} PRIVATE dawnwood_core CUDA::cudart)
    set_target_properties(${target} PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON CUDA_RUNTIME_LIBRARY Shared)
    target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;-lineinfo;--ptxas-options=-v>)
  endforeach()
  target_compile_definitions(dawnwood_reference PRIVATE DW_REFERENCE_KERNEL=1)
  foreach(target dawnwood_integer dawnwood_integer_texture dawnwood_integer_reference dawnwood_integer_cached)
    add_executable(${target} src/integer_main.cu)
    target_link_libraries(${target} PRIVATE dawnwood_core CUDA::cudart)
    set_target_properties(${target} PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON CUDA_RUNTIME_LIBRARY Shared)
    target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;-lineinfo;--ptxas-options=-v>)
  endforeach()
  target_compile_definitions(dawnwood_integer_texture PRIVATE DWI_DIRECT_TEXTURE=1)
  target_compile_definitions(dawnwood_integer_reference PRIVATE DWI_MASKED=0)
  target_compile_definitions(dawnwood_integer_cached PRIVATE DWI_DIRECT_TEXTURE=1 DWI_PAIR_CACHE=1)
  add_test(NAME dawnwood_integer_cached_gpu COMMAND dawnwood_integer_cached --self-test)
  add_test(NAME dawnwood_integer_gpu COMMAND dawnwood_integer --self-test)
  add_test(NAME dawnwood_integer_texture_gpu COMMAND dawnwood_integer_texture --self-test)
  add_test(NAME dawnwood_integer_reference_gpu COMMAND dawnwood_integer_reference --self-test)
  set_tests_properties(dawnwood_integer_gpu dawnwood_integer_texture_gpu dawnwood_integer_reference_gpu dawnwood_integer_cached_gpu PROPERTIES LABELS "gpu;integer")
  add_test(NAME dawnwood_gpu COMMAND dawnwood --self-test)
  set_tests_properties(dawnwood_gpu PROPERTIES LABELS "gpu")
endif()
