cmake_minimum_required(VERSION 3.22)
project(DawnwoodGPU VERSION 0.5.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

# Refuse stale CPU/shader equation pairs even on an Android build without Python.
set(DW_SHADER_HASH_INPUT "")
foreach(DW_SOURCE include/numeric_types.inc include/numeric_math.inc include/numeric_evolve.inc shaders/mutate.comp shaders/evolve.comp shaders/operator_texture.inc shaders/state_buffers.inc shaders/evolution_pass.inc shaders/prepare.comp shaders/slope.comp shaders/combine.comp shaders/geometry.comp shaders/finish.comp)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${DW_SOURCE}" DW_PART_HASH)
  string(APPEND DW_SHADER_HASH_INPUT "${DW_PART_HASH}")
endforeach()
string(SHA256 DW_CURRENT_SHADER_HASH "${DW_SHADER_HASH_INPUT}")
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/source.sha256")
  message(FATAL_ERROR "Run python tools/build_shaders.py before configuring.")
endif()
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/source.sha256" DW_SHIPPED_SHADER_HASH LIMIT_COUNT 1)
if(NOT DW_CURRENT_SHADER_HASH STREQUAL DW_SHIPPED_SHADER_HASH)
  message(FATAL_ERROR "Numerical/shader sources changed. Run python tools/build_shaders.py, then configure again.")
endif()

if(ANDROID)
  add_library(dawnwood_vulkan INTERFACE)
  target_link_libraries(dawnwood_vulkan INTERFACE vulkan)
  add_library(Vulkan::Vulkan ALIAS dawnwood_vulkan)
else()
  find_package(Vulkan REQUIRED)
endif()
add_library(dawnwood_core STATIC src/cpu.cpp src/vulkan.cpp src/driver.cpp)
target_include_directories(dawnwood_core PUBLIC include PRIVATE src)
target_link_libraries(dawnwood_core PUBLIC Vulkan::Vulkan)
if(MSVC)
  target_compile_options(dawnwood_core PRIVATE /W4 /fp:strict)
else()
  target_compile_options(dawnwood_core PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
if(ANDROID)
  add_library(dawnwood_jni SHARED android/app/src/main/cpp/jni.cpp)
  target_link_libraries(dawnwood_jni PRIVATE dawnwood_core log)
else()
  add_executable(dawnwood src/main.cpp)
  target_link_libraries(dawnwood PRIVATE dawnwood_core)
  enable_testing()
  add_test(NAME cpu_source_profile COMMAND dawnwood selftest)
endif()
